#define trigPin 8
#define echoPin 9
#define tempPin A1

int gasPin = A5;
int led = 12;
int buzzer = A0;

int gas;
float temp, distance;

bool gasAlert = false;
bool tempAlert = false;
bool distAlert = false;
bool alert = false;

bool monitoring = false;
bool programExit = false;

void setup() {
  Serial.begin(9600);

  pinMode(led, OUTPUT);
  pinMode(buzzer, OUTPUT);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  showMenu();
}

void loop() {
  if (programExit)
    return;

  if (monitoring)
    automaticMonitoring();

  if (Serial.available())
    executeCommand();
}

//---------------- MENU ----------------

void showMenu() {
  Serial.println();
  Serial.println("====== SMART CPS ======");
  Serial.println("1. View Gas Status");
  Serial.println("2. View Temperature");
  Serial.println("3. View Distance");
  Serial.println("4. View All Sensors");
  Serial.println("5. LED ON");
  Serial.println("6. LED OFF");
  Serial.println("7. Buzzer ON");
  Serial.println("8. Buzzer OFF");
  Serial.println("9. Automatic Monitoring");
  Serial.println("0. Exit Monitoring");
  Serial.print("\nEnter Choice: ");
}

//---------------- READ SENSORS ----------------

void readSensors() {

  // Read Gas Sensor
  gas = analogRead(gasPin);

  // Read LM35 Temperature Sensor
  int tempValue = analogRead(tempPin);

  float voltage = tempValue * (5.0 / 1023.0);

  temp = voltage * 100.0;

  // Read Ultrasonic Sensor
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH);

  distance = duration * 0.034 / 2;
}

//---------------- CHECK STATUS ----------------

void checkStatus() {

  gasAlert = false;
  tempAlert = false;
  distAlert = false;
  alert = false;

  if (gas > 939) {
    gasAlert = true;
    alert = true;
  }

  if (temp > 35) {
    tempAlert = true;
    alert = true;
  }

  if (distance < 15) {
    distAlert = true;
    alert = true;
  }
}

//---------------- CONTROL OUTPUTS ----------------

void controlOutputs() {

  if (alert) {
    digitalWrite(led, HIGH);
    tone(buzzer, 1000);
  }
  else {
    digitalWrite(led, LOW);
    noTone(buzzer);
  }
}

//---------------- GAS ----------------

void showGas() {

  readSensors();
  checkStatus();
  controlOutputs();

  Serial.println("\n------ GAS SENSOR ------");

  Serial.print("Gas Value : ");
  Serial.println(gas);

  if (gasAlert)
    Serial.println("Status : ALERT");
  else
    Serial.println("Status : SAFE");
}

//---------------- TEMPERATURE ----------------

void showTemp() {

  readSensors();
  checkStatus();
  controlOutputs();

  Serial.println("\n---- TEMPERATURE ----");

  Serial.print("Temperature : ");
  Serial.print(temp);
  Serial.println(" C");

  if (tempAlert)
    Serial.println("Status : HIGH");
  else
    Serial.println("Status : NORMAL");
}

//---------------- DISTANCE ----------------

void showDistance() {

  readSensors();
  checkStatus();
  controlOutputs();

  Serial.println("\n------ DISTANCE ------");

  Serial.print("Distance : ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distAlert)
    Serial.println("Status : OBJECT DETECTED");
  else
    Serial.println("Status : CLEAR");
}

//---------------- DASHBOARD ----------------

void dashboard() {

  Serial.println();
  Serial.println("====== SMART CPS ======");

  Serial.print("Gas Value : ");
  Serial.println(gas);

  if (gasAlert)
    Serial.println("Gas Status : ALERT");
  else
    Serial.println("Gas Status : SAFE");

  Serial.print("Temperature : ");
  Serial.print(temp);
  Serial.println(" C");

  if (tempAlert)
    Serial.println("Temp Status : HIGH");
  else
    Serial.println("Temp Status : NORMAL");

  Serial.print("Distance : ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distAlert)
    Serial.println("Distance : OBJECT DETECTED");
  else
    Serial.println("Distance : CLEAR");

  Serial.println("========================");
}

//---------------- AUTOMATIC MONITORING ----------------

void automaticMonitoring() {

  readSensors();
  checkStatus();
  controlOutputs();
  dashboard();

  Serial.println("Press 0 to Stop Monitoring");

  unsigned long start = millis();

  while (millis() - start < 1000) {

    if (Serial.available()) {

      String input = Serial.readStringUntil('\n');
      input.trim();

      if (input == "0") {

        monitoring = false;

        noTone(buzzer);
        digitalWrite(led, LOW);

        Serial.println("Monitoring Stopped");

        showMenu();
        return;
      }
    }
  }
}

//---------------- COMMANDS ----------------

void executeCommand() {

  String input = Serial.readStringUntil('\n');
  input.trim();

  if (input.length() == 0)
    return;

  Serial.println(input);

  switch (input.charAt(0)) {

    case '1':
      showGas();
      break;

    case '2':
      showTemp();
      break;

    case '3':
      showDistance();
      break;

    case '4':
      readSensors();
      checkStatus();
      controlOutputs();
      dashboard();
      break;

    case '5':
      digitalWrite(led, HIGH);
      Serial.println("LED ON");
      break;

    case '6':
      digitalWrite(led, LOW);
      Serial.println("LED OFF");
      break;

    case '7':
      tone(buzzer, 1000);
      Serial.println("BUZZER ON");
      break;

    case '8':
      noTone(buzzer);
      Serial.println("BUZZER OFF");
      break;

    case '9':
      monitoring = true;
      Serial.println("Automatic Monitoring Started");
      return;

    case '0':
      monitoring = false;
      programExit = true;

      noTone(buzzer);
      digitalWrite(led, LOW);

      Serial.println("Program Closed");
      return;

    default:
      Serial.println("INVALID CHOICE");
  }

  showMenu();
}
