#define PIR_PIN 13 
#define TRIG_PIN 8 
#define ECHO_PIN 9 
#define LED_PIN 12 
#define BUZZER A0 
String username = ""; 
String password = ""; 
int attempts = 0; 
void setup() { 
  Serial.begin(9600); 
  pinMode(PIR_PIN, INPUT); 
  pinMode(TRIG_PIN, OUTPUT); 
  pinMode(ECHO_PIN, INPUT); 
  pinMode(LED_PIN, OUTPUT); 
  pinMode(BUZZER, OUTPUT); 
  Serial.println("================================"); 
  Serial.println("      IIoT SECURITY SYSTEM"); 
  Serial.println("================================"); 
} 
void loop() { 
  Serial.println("\nWaiting for Motion..."); 
  while (digitalRead(PIR_PIN) == LOW); 
  float distance = getDistance(); 
  Serial.print("Distance : "); 
  Serial.print(distance); 
  Serial.println(" cm"); 
  if (distance > 50) { 
    Serial.println("Object Too Far"); 
    delay(1000); 
    return; 
  } 
  login(); 
  Serial.println("\nMonitoring Area..."); 
  delay(3000); 
} 
//--------------- LOGIN ---------------- 
void login() { 
  while (true) { 
    Serial.print("Enter Username:"); 
    while (!Serial.available()); 
    username = Serial.readStringUntil('\n'); 
    username.trim(); 
    Serial.println(username); 
    if (username == "User" || username == "Admin" || username == "System") { 
      Serial.println("Username Accepted"); 
      blinkOnce(); 
      break; 
    } else { 
      Serial.println("INVALID USERNAME"); 
      lockSystem(); 
    } 
  } 
  attempts = 0; 
  while (attempts < 3) { 
    Serial.print("Enter Password:"); 
    while (!Serial.available()); 
    password = Serial.readStringUntil('\n'); 
    password.trim(); 
    Serial.println(password); 
    if (password == "1234") { 
      Serial.println("ACCESS GRANTED"); 
      blinkTwice(); 
      return; 
    } else { 
      attempts++; 
      Serial.print("Wrong Password"); 
      if (attempts < 3) { 
        Serial.print("  Attempts Left : "); 
        Serial.println(3 - attempts); 
      } 
    } 
  } 
  lockSystem(); 
} 
//--------------- DISTANCE ---------------- 
float getDistance() { 
  digitalWrite(TRIG_PIN, LOW); 
  delayMicroseconds(2); 
  digitalWrite(TRIG_PIN, HIGH); 
  delayMicroseconds(10); 
  digitalWrite(TRIG_PIN, LOW); 
  long duration = pulseIn(ECHO_PIN, HIGH); 
  return duration * 0.034 / 2; 
} 
//--------------- LED BLINKS ---------------- 
void blinkOnce() { 
  digitalWrite(LED_PIN, HIGH); 
  delay(300); 
  digitalWrite(LED_PIN, LOW); 
} 
void blinkTwice() { 
  for (int i = 0; i < 2; i++) { 
    digitalWrite(LED_PIN, HIGH); 
    delay(300); 
    digitalWrite(LED_PIN, LOW); 
    delay(300); 
  } 
} 
//--------------- SYSTEM LOCK ---------------- 
void lockSystem() { 
  Serial.println(); 
  Serial.println("************************"); 
  Serial.println(" SECURITY BREACH"); 
  Serial.println(" SYSTEM LOCKED"); 
  Serial.println(" RESET REQUIRED"); 
  Serial.println("************************"); 
  while (true) { 
    digitalWrite(LED_PIN, HIGH); 
    tone(BUZZER, 1000); 
    delay(300); 
    digitalWrite(LED_PIN, LOW); 
    noTone(BUZZER); 
    delay(300); 
  } 
}
